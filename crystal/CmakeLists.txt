cmake_minimum_required(VERSION 3.31.0)
project(crystal VERSION 0.1.0 LANGUAGES C CXX)

##################################################
#Setting up build options
##################################################
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

set(CRYSTAL_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

# set source files based on folders
set(CRYSTAL_CORE_SRC
        src/core/Window.cpp
        src/core/Logger.cpp
        src/core/Engine.cpp)
set(CRYSTAL_GEOMETRY_SRC
        src/geometry/Matrix.cpp
        src/geometry/Vector3.cpp
        src/layout/Size.cpp)
set(CRYSTAL_UI_SRC)

set(CRYSTAL_GRAPHICS_SRC
        src/graphics/Color.cpp
        src/graphics/Buffer.cpp
        src/graphics/Graphics.cpp
        src/graphics/Shader.cpp
        src/graphics/VertexArray.cpp
        src/graphics/Renderer.cpp)
set(CRYSTAL_GL_GRAPHICS_SRC
        src/graphics/GL/GLRenderer.cpp
        src/graphics/GL/GLHelper.cpp
        src/graphics/GL/GLShader.cpp)

# reference all source files in one variable
set(CRYSTAL_SOURCE_FILES
        ${CRYSTAL_CORE_SRC}
        ${CRYSTAL_GEOMETRY_SRC}
        ${CRYSTAL_UI_SRC}
        ${CRYSTAL_GRAPHICS_SRC}
        ${CRYSTAL_GL_GRAPHICS_SRC})

# set include directories
set(CRYSTAL_INCLUDE_DIRECTORIES ${CRYSTAL_DIRECTORY}/include)


add_subdirectory(external/glfw)
add_subdirectory(external/glad)
find_package(OpenGL REQUIRED)


# build library
add_library(crystal STATIC ${CRYSTAL_SOURCE_FILES})
target_compile_options(crystal PRIVATE
        "-pedantic"
        "-Wfloat-equal"
        "-Wundef"
        "-Wpointer-arith"
        "-Wwrite-strings"
        "-Wswitch-default"
        "-Wswitch-enum")

# include
target_include_directories(crystal PUBLIC ${CRYSTAL_INCLUDE_DIRECTORIES})

# linking
target_link_libraries(crystal PRIVATE glfw OpenGL::GL glad dwmapi)
